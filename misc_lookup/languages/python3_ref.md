# Basic Interpolation & Self-Documenting (Python 3.8+)

print(f"{'test':^10}")                    # '   test   ' (Centered)
print(f"{'test':*^10}")                   # '***test***' (Fill character)
print(f"{42:05d}")                        # '00042' (Zero padded)

# Base Conversions
n = 255
print(f"{n:b} | {n:o} | {n:x} | {n:X}")   # 11111111 | 377 | ff | FF
